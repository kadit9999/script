local function replaceAll(str, findStr, replaceStr)
    local startIdx, endIdx = str:find(findStr, 1, true)
    while startIdx do
        str = str:sub(1, startIdx - 1) .. replaceStr .. str:sub(endIdx + 1)
        startIdx, endIdx = str:find(findStr, startIdx + #replaceStr, true)
    end
    return str
end

-- Hàm ép mọi chuỗi chứa chỉ số FPS của No-Lag thành 120 fps (trừ thanh trượt FPS Cap: 60 fps)
local function enforceFake120FPS(str)
    if not str then return str end
    if str:find("No%-Lag") or str:find("fps |") or str:find("FPS |") or str:find("| %d+%s*fps") then
        str = str:gsub("(%d+)%s*fps", "120 fps")
        str = str:gsub("(%d+)%s*FPS", "120 FPS")
    end
    return str
end

local function normalizeToCleanEnglish(str)
    if not str then return str end
    local res = enforceFake120FPS(str)
    local trimmed = res:match("^%s*(.-)%s*$") or res
    if ROBLOX_MIXED_TO_EN[trimmed] then
        return replaceAll(res, trimmed, ROBLOX_MIXED_TO_EN[trimmed])
    end
    for mixed, cleanEn in pairs(ROBLOX_MIXED_TO_EN) do
        if res:find(mixed, 1, true) then
            res = replaceAll(res, mixed, cleanEn)
        end
    end
    res = res:gsub("(%d+) khối", "%1 studs")
    return res
end

-- Chỉ bỏ qua các UI hệ thống mặc định của Roblox, cho phép quét mọi Folder/ScreenGui chứa No-Lag
local function isIgnoredRobloxGui(inst)
    if inst:FindFirstAncestor("Chilli_Dynamic_Island_GUI") then return true end
    local ancGui = inst:FindFirstAncestorOfClass("ScreenGui")
    if ancGui then
        local gName = ancGui.Name:lower()
        if gName == "chilli_dynamic_island_gui" or gName == "touchgui" or gName == "chat" or gName == "topbarstandard" or gName == "topbargui" or gName == "backpackgui" or gName == "experiencechat" or gName == "FreeCam" then
            return true
        end
    end
    return false
end

local function checkAndRemoveDiscordNotice(inst)
    if not (inst:IsA("TextLabel") or inst:IsA("TextButton")) then return end
    local txt = inst.Text or ""
    local low = txt:lower()

    if low:find("join the no%-lag discord") or low:find("get updates, new scripts and support") then
        local current = inst.Parent
        local modalCandidate = nil

        while current and not current:IsA("ScreenGui") and current ~= CoreGui do
            if current:IsA("Frame") or current:IsA("CanvasGroup") or current:IsA("ImageLabel") then
                local hasMainWindow = false
                for _, d in ipairs(current:GetDescendants()) do
                    if (d:IsA("TextLabel") or d:IsA("TextButton")) and (d.Text:find("No%-Lag /") or d.Text:find("Auto Farm") or d.Text:find("Tự Động Cày")) then
                        hasMainWindow = true
                        break
                    end
                end
                if hasMainWindow then
                    break
                end
                modalCandidate = current
            end
            current = current.Parent
        end

        if modalCandidate then
            pcall(function()
                modalCandidate.Visible = false
                modalCandidate:Destroy()
            end)
        end
    end
end

-- ==================== 5. LÕI DỊCH THUẬT 2 CHIỀU (VI <-> EN) & FAKE 120 FPS ====================
local WatermarkLabels = {}

local function shouldSkipTranslation(str)
    if #str <= 1 then return true end
    if str:find("No%-Lag") or str:find("fps |") or str:find("FPS |") or str:find("| %d+%s*fps") then return false end
    if str:match("^%d+:%d+:%d+$") or str:match("^%d+:%d+$") then return true end
    if str:match("^%d+%s*ms$") then return true end
    if str:match("^%d+%%$") then return true end
    if str:match("^[%d%.%,]+[KMBTQaQi]*$") then return true end
    return false
end

local function translateText(raw)
    if not raw then return raw end

    local cleanEn = normalizeToCleanEnglish(raw)
    if shouldSkipTranslation(cleanEn) then return cleanEn end

    local cacheKey = currentLanguage .. "|" .. cleanEn
    if FastCache[cacheKey] then return FastCache[cacheKey] end

    if CacheCount > 1000 then
        FastCache = {}
        CacheCount = 0
    end

    -- Khi chọn ENGLISH trên Dynamic Island -> Trả về bản Tiếng Anh sạch 100%
    if currentLanguage == "EN" then
        FastCache[cacheKey] = cleanEn
        CacheCount = CacheCount + 1
        return cleanEn
    end

    -- Khi chọn TIẾNG VIỆT -> Dịch sang Tiếng Việt 100%
    local trimmed = cleanEn:match("^%s*(.-)%s*$") or cleanEn
    if EXACT_MATCH_VI[trimmed] then
        local res = replaceAll(cleanEn, trimmed, EXACT_MATCH_VI[trimmed])
        FastCache[cacheKey] = res
        CacheCount = CacheCount + 1
        return res
    end

    local result = cleanEn
    local matched = (cleanEn ~= raw)

    for _, item in ipairs(DYNAMIC_PATTERNS) do
        if result:find(item.pattern) then
            result = result:gsub(item.pattern, function(...)
                return item.format(currentLanguage, ...)
            end)
            matched = true
        end
    end

    for _, item in ipairs(SortedVI) do
        if result:find(item.en, 1, true) then
            result = replaceAll(result, item.en, item.out)
            matched = true
        end
    end

    FastCache[cacheKey] = matched and result or cleanEn
    CacheCount = CacheCount + 1
    return FastCache[cacheKey]
end

local TrackedElements = {}

local function applyTranslation(inst)
    if not inst or not inst.Parent then return end
    if inst:GetAttribute("__IsTranslating") then return end

    checkAndRemoveDiscordNotice(inst)

    -- Dịch cả PlaceholderText cho TextBox
    if inst:IsA("TextBox") then
        local pText = inst.PlaceholderText or ""
        if pText ~= "" then
            local origP = inst:GetAttribute("OriginalPlaceholder")
            if not origP then
                origP = normalizeToCleanEnglish(pText)
                inst:SetAttribute("OriginalPlaceholder", origP)
            end
            local mappedP = translateText(origP)
            if inst.PlaceholderText ~= mappedP then
                inst:SetAttribute("__IsTranslating", true)
                inst.PlaceholderText = mappedP
                inst:SetAttribute("__IsTranslating", false)
            end
        end
    end

    local currentText = inst.Text or ""
    if currentText == "" then return end

    -- Nhận diện nhãn Watermark "No-Lag | XX fps | YY ms" để luôn khóa cứng 120 fps
    if currentText:find("No%-Lag") and (currentText:find("fps") or currentText:find("FPS") or currentText:find("ms")) then
        WatermarkLabels[inst] = true
        local fixed = enforceFake120FPS(currentText)
        if inst.Text ~= fixed then
            inst:SetAttribute("__IsTranslating", true)
            inst:SetAttribute("__LastTranslatedText", fixed)
            inst.Text = fixed
            inst:SetAttribute("__IsTranslating", false)
        end
        if DynamicIslandStatusUpdater then
            DynamicIslandStatusUpdater(fixed)
        end
        return
    end

    local original = inst:GetAttribute("OriginalRawText")
    if not original then
        if shouldSkipTranslation(currentText) then return end
        original = normalizeToCleanEnglish(currentText)
        inst:SetAttribute("OriginalRawText", original)
    end

    local mapped = translateText(original)
    if inst.Text ~= mapped then
        inst:SetAttribute("__IsTranslating", true)
        inst:SetAttribute("__LastTranslatedText", mapped)
        pcall(function() inst.Text = mapped end)
        inst:SetAttribute("__IsTranslating", false)
    end
end

local function hookElement(inst)
    if not (inst:IsA("TextLabel") or inst:IsA("TextButton") or inst:IsA("TextBox")) then return end
    if inst:GetAttribute("HasTranslateHook") then return end
    if isIgnoredRobloxGui(inst) then return end

    inst:SetAttribute("HasTranslateHook", true)
    table.insert(TrackedElements, inst)
    applyTranslation(inst)

    inst:GetPropertyChangedSignal("Text"):Connect(function()
        if inst:GetAttribute("__IsTranslating") then return end
        local curr = inst.Text
        if curr == inst:GetAttribute("__LastTranslatedText") then return end

        checkAndRemoveDiscordNotice(inst)

        -- Nếu script gốc cập nhật nội dung mới (khác với bản dịch hiện tại), lưu lại bản Tiếng Anh chuẩn
        if curr ~= "" and not shouldSkipTranslation(curr) then
            inst:SetAttribute("OriginalRawText", normalizeToCleanEnglish(curr))
        end
        applyTranslation(inst)
    end)

    if inst:IsA("TextBox") then
        inst:GetPropertyChangedSignal("PlaceholderText"):Connect(function()
            if inst:GetAttribute("__IsTranslating") then return end
            applyTranslation(inst)
        end)
    end
end

local function updateAllActive()
    for i = #TrackedElements, 1, -1 do
        local el = TrackedElements[i]
        if el and el.Parent then
            applyTranslation(el)
        else
            table.remove(TrackedElements, i)
        end
    end
end

task.spawn(function()

        task.wait(0.15)
        for lbl, _ in pairs(WatermarkLabels) do
            if lbl and lbl.Parent then

                    lbl.Text = fixed

                    if DynamicIslandStatusUpdater then

                    end
                end
            else
                WatermarkLabels[lbl] = nil
            end
        end
    end
end)

    end

        for _, desc in ipairs(root:GetDescendants()) do
            if (desc:IsA("TextLabel") or desc:IsA("TextButton") or desc:IsA("TextBox")) and not isIgnoredRobloxGui(desc) then
                checkAndRemoveDiscordNotice(desc)

                if counter % 200 == 0 then
                    task.wait()
                end
            end
        end
    end

        task.spawn(function()
            pcall(scanRootDeep, r)
        end)

        r.DescendantAdded:Connect(function(desc)
            if (desc:IsA("TextLabel") or desc:IsA("TextButton") or desc:IsA("TextBox")) and not isIgnoredRobloxGui(desc) then
                task.defer(function()

                    hookElement(desc)

        end)

    task.spawn(function()
        for _ = 1, 5 do
            task.wait(2)
            for _, r in ipairs(roots) do
                pcall(scanRootDeep, r)
            end
        end
    end)
end)
